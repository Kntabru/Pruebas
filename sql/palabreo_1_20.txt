-- Palabreo 1.20: migration for the observed existing database, NOT a fresh schema.
-- DO NOT RUN until the complete script and backups have been reviewed.
-- Preserves users, profiles, leagues, challenges and historical results.
-- Changes only word lengths and adds the 6/7-letter catalog. No rows are deleted.
-- Requires a privileged SQL Editor session. All changes are atomic.
BEGIN;
SET LOCAL lock_timeout = '10s';
SET LOCAL statement_timeout = '120s';
SELECT pg_advisory_xact_lock(hashtext('palabreo-migration-1.20'));
DO $guard$
BEGIN
 IF to_regclass('palabreo_private.word_catalog') IS NULL
 OR to_regprocedure('public.palabreo_guess(uuid,text,uuid)') IS NULL
 THEN RAISE EXCEPTION 'Missing expected Palabreo schema. Nothing applied.'; END IF;
 IF EXISTS(SELECT 1 FROM pg_attribute WHERE attrelid='public.leagues'::regclass AND attname='length_mode' AND NOT attisdropped)
 THEN RAISE EXCEPTION 'Length migration already present or schema differs. Do not run again.'; END IF;
END $guard$;
ALTER TABLE public.leagues ADD COLUMN length_mode text NOT NULL DEFAULT '5'
 CHECK (length_mode IN ('5','6','7','random'));
ALTER TABLE palabreo_private.daily_plans ADD COLUMN word_length smallint NOT NULL DEFAULT 5
 CHECK (word_length IN (5,6,7));
ALTER TABLE public.challenges ADD COLUMN word_length smallint NOT NULL DEFAULT 5
 CHECK (word_length IN (5,6,7));
ALTER TABLE public.challenges ADD CONSTRAINT challenges_global_five_letters
 CHECK (league_id IS NOT NULL OR word_length=5);
-- Defaults retain every existing league, day and challenge at five letters.
ALTER TABLE palabreo_private.word_catalog DROP CONSTRAINT word_catalog_word_check;
ALTER TABLE palabreo_private.word_catalog ADD CONSTRAINT word_catalog_word_check CHECK (word ~ '^[A-ZÑ]{5,7}$');
ALTER TABLE palabreo_private.solutions DROP CONSTRAINT solutions_answer_check;
ALTER TABLE palabreo_private.solutions ADD CONSTRAINT solutions_answer_check CHECK (answer ~ '^[A-ZÑ]{5,7}$');
ALTER TABLE palabreo_private.daily_word_reservations DROP CONSTRAINT daily_word_reservations_answer_check;
ALTER TABLE palabreo_private.daily_word_reservations ADD CONSTRAINT daily_word_reservations_answer_check CHECK (answer ~ '^[A-ZÑ]{5,7}$');
ALTER TABLE palabreo_private.guesses DROP CONSTRAINT guesses_guess_check;
ALTER TABLE palabreo_private.guesses ADD CONSTRAINT guesses_guess_check CHECK (guess ~ '^[A-ZÑ]{5,7}$');
ALTER TABLE palabreo_private.guesses DROP CONSTRAINT guesses_clues_check;
ALTER TABLE palabreo_private.guesses ADD CONSTRAINT guesses_clues_check
 CHECK (jsonb_typeof(clues)='array' AND jsonb_array_length(clues)=char_length(guess));
CREATE INDEX word_catalog_length_answers_idx ON palabreo_private.word_catalog ((char_length(word)),is_answer);
CREATE FUNCTION palabreo_private.freeze_league_length() RETURNS trigger
LANGUAGE plpgsql SET search_path TO '' AS $body$
BEGIN
 IF NEW.length_mode IS DISTINCT FROM OLD.length_mode THEN
  RAISE EXCEPTION 'La longitud de una liga no se puede cambiar después de crearla' USING ERRCODE='22023';
 END IF;
 RETURN NEW;
END $body$;
REVOKE ALL ON FUNCTION palabreo_private.freeze_league_length() FROM PUBLIC,anon,authenticated;
CREATE TRIGGER freeze_league_length BEFORE UPDATE ON public.leagues
 FOR EACH ROW EXECUTE FUNCTION palabreo_private.freeze_league_length();

CREATE OR REPLACE FUNCTION public.palabreo_create_league_v120(p_name text, p_daily_words integer, p_length_mode text)
 RETURNS uuid
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$
declare lid uuid;
begin
 perform palabreo_private.require_scope(null);
 perform 1 from public.profiles where id=auth.uid() for update;
 if length(trim(p_name)) not between 3 and 60 then raise exception 'El nombre necesita entre 3 y 60 caracteres';end if;
 if (select count(*) from public.league_members where user_id=auth.uid() and left_at is null)>=5 then raise exception 'Puedes participar en un máximo de cinco ligas privadas, contando las creadas y las que te unes';end if;
 if p_daily_words is null or p_daily_words not between 3 and 9 then raise exception 'Elige entre tres y nueve palabras diarias';end if;
 if p_length_mode IS NULL OR p_length_mode NOT IN ('5','6','7','random') then raise exception 'Elige longitud 5, 6, 7 o aleatoria' using errcode='22023';end if;
 insert into public.leagues(name,owner_id,daily_words,length_mode) values(trim(p_name),auth.uid(),p_daily_words,p_length_mode) returning id into lid;
 insert into public.league_members(league_id,user_id) values(lid,auth.uid());return lid;
end $function$
;

REVOKE ALL ON FUNCTION public.palabreo_create_league_v120(text,integer,text) FROM PUBLIC,anon;
GRANT EXECUTE ON FUNCTION public.palabreo_create_league_v120(text,integer,text) TO authenticated;

CREATE OR REPLACE FUNCTION palabreo_private.assign_day(p_league uuid)
 RETURNS void
 LANGUAGE plpgsql
 SET search_path TO ''
AS $function$
declare d date:=palabreo_private.today();v_slot integer;chosen text;cid uuid;kind text;day_count integer;length_mode text;chosen_length smallint;
begin
 -- Serializa la asignación diaria para evitar coincidencias entre ámbitos.
 perform pg_advisory_xact_lock(hashtext('palabreo-day:'||d::text));
 if p_league is not null then perform palabreo_private.assign_day(null);end if;
 kind:=case when p_league is null then 'global' else 'private' end;
 if p_league is null then day_count:=5;chosen_length:=5;
 else
  select l.daily_words,l.length_mode into day_count,length_mode from public.leagues l where id=p_league;
  if not found then raise exception 'La liga no existe';end if;
  chosen_length:=case when length_mode='random' then (5+floor(random()*3))::smallint else length_mode::smallint end;
  insert into palabreo_private.daily_plans(scope_key,play_date,daily_words,word_length) values(p_league::text,d,day_count,chosen_length) on conflict do nothing;
  select dp.daily_words,dp.word_length into day_count,chosen_length from palabreo_private.daily_plans dp where scope_key=p_league::text and play_date=d;
 end if;
 for v_slot in 1..day_count loop
  if not exists(select 1 from public.challenges c join palabreo_private.solutions s on s.challenge_id=c.id where c.play_date=d and c.slot=v_slot and c.league_id is not distinct from p_league) then
   select w.word into chosen from palabreo_private.word_catalog w
   where w.is_answer and char_length(w.word)=chosen_length
    and not exists(select 1 from palabreo_private.daily_word_reservations x where x.play_date=d and x.answer=w.word and x.competition_kind<>kind)
    and not exists(select 1 from public.challenges c join palabreo_private.solutions s on s.challenge_id=c.id where c.play_date between d-7 and d and c.league_id is not distinct from p_league and s.answer=w.word)
   order by random() limit 1;
   if chosen is null then raise exception 'No hay suficientes palabras disponibles';end if;
   insert into public.challenges(league_id,play_date,slot,word_length) values(p_league,d,v_slot,chosen_length)
    on conflict do nothing;
   select c.id into cid from public.challenges c where c.league_id is not distinct from p_league and c.play_date=d and c.slot=v_slot;
   insert into palabreo_private.solutions(challenge_id,answer) values(cid,chosen);
  end if;
 end loop;
end $function$
;

CREATE OR REPLACE FUNCTION palabreo_private.freeze_challenge()
 RETURNS trigger
 LANGUAGE plpgsql
 SET search_path TO ''
AS $function$
begin
 if new.play_date<>old.play_date
  or new.league_id is distinct from old.league_id
  or new.slot<>old.slot
  or new.word_length<>old.word_length then
  raise exception 'No se puede cambiar la fecha ni el ámbito de un desafío';
 end if;
 return new;
end $function$
;

CREATE OR REPLACE FUNCTION palabreo_private.reserve_daily_word()
 RETURNS trigger
 LANGUAGE plpgsql
 SET search_path TO ''
AS $function$
declare
 challenge public.challenges%rowtype;
begin
 if tg_op='UPDATE' then
  raise exception 'Las soluciones asignadas son inmutables';
 end if;

 select * into strict challenge
 from public.challenges
 where id=new.challenge_id;

 if char_length(new.answer)<>challenge.word_length then raise exception 'La solución no coincide con la longitud del desafío';end if;
 new.play_date:=challenge.play_date;
 new.competition_kind:=
  case when challenge.league_id is null then 'global' else 'private' end;
 new.scope_key:=
  coalesce(challenge.league_id,'00000000-0000-0000-0000-000000000000'::uuid);

 insert into palabreo_private.daily_word_reservations(
  play_date,answer,competition_kind
 )
 values(new.play_date,new.answer,new.competition_kind)
 on conflict(play_date,answer) do nothing;

 return new;
end $function$
;

CREATE OR REPLACE FUNCTION palabreo_private.round_view(p_round uuid)
 RETURNS jsonb
 LANGUAGE sql
 STABLE
 SET search_path TO ''
AS $function$
 select jsonb_build_object('id',r.id,'challenge_id',r.challenge_id,'status',r.status,'attempts',r.attempts,'points',r.points,
 'started_at',r.started_at,'finished_at',r.finished_at,'duration_ms',r.duration_ms,'server_time',clock_timestamp(),
 'word_length',c.word_length,'date',c.play_date,'slot',c.slot,'league_id',c.league_id,'daily_words',(select count(*) from public.challenges day_c where day_c.play_date=c.play_date and day_c.league_id is not distinct from c.league_id),
 'guesses',coalesce((select jsonb_agg(jsonb_build_object('word',g.guess,'result',g.clues) order by g.attempt_number) from palabreo_private.guesses g where g.round_id=r.id),'[]'::jsonb),
 'answer',case when r.status<>'playing' then s.answer else null end)
 from public.competition_rounds r join public.challenges c on c.id=r.challenge_id join palabreo_private.solutions s on s.challenge_id=c.id where r.id=p_round
$function$
;

CREATE OR REPLACE FUNCTION public.palabreo_day(p_league uuid DEFAULT NULL::uuid)
 RETURNS jsonb
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$
declare result jsonb;
begin
 perform palabreo_private.require_scope(p_league);perform palabreo_private.assign_day(p_league);perform palabreo_private.expire_rounds();
 select jsonb_build_object('date',palabreo_private.today(),'week',public.palabreo_week(palabreo_private.today()),
 'daily_words',(select count(*) from public.challenges dc where dc.play_date=palabreo_private.today() and dc.league_id is not distinct from p_league),'league_id',p_league,'name',coalesce((select l.name from public.leagues l where l.id=p_league),'Competición global'),
 'word_length',coalesce(max(c.word_length),5),'challenges',coalesce(jsonb_agg(jsonb_build_object('id',c.id,'word_length',c.word_length,'slot',c.slot,'status',coalesce(r.status,'new'),'attempts',coalesce(r.attempts,0),'points',coalesce(r.points,0)) order by c.slot),'[]'::jsonb)) into result
 from public.challenges c left join public.competition_rounds r on r.challenge_id=c.id and r.user_id=auth.uid()
 where c.play_date=palabreo_private.today() and c.league_id is not distinct from p_league;
 return result;
end $function$
;

CREATE OR REPLACE FUNCTION public.palabreo_guess(p_round uuid, p_guess text, p_request_id uuid)
 RETURNS jsonb
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$
declare r public.competition_rounds%rowtype;c public.challenges%rowtype;solution text;guess_word text;remaining text[];clues text[];i integer;j integer;ended timestamptz:=clock_timestamp();new_status text;
begin
 select * into strict r from public.competition_rounds where id=p_round for update;
 if r.user_id is distinct from auth.uid() then raise exception 'Esta ronda no te pertenece' using errcode='42501';end if;
 select * into strict c from public.challenges where id=r.challenge_id;perform palabreo_private.require_scope(c.league_id);
 if p_request_id is null then raise exception 'Falta el identificador del intento';end if;
 if exists(select 1 from palabreo_private.guesses g where g.round_id=r.id and g.request_id=p_request_id) then return palabreo_private.round_view(r.id);end if;
 if r.status<>'playing' then return palabreo_private.round_view(r.id);end if;
 if c.play_date<>palabreo_private.today() then
  update public.competition_rounds set status='abandoned',finished_at=((c.play_date+1)::timestamp at time zone 'Europe/Madrid') where id=r.id;
  return palabreo_private.round_view(r.id);
 end if;
 guess_word:=upper(translate(lower(trim(p_guess)),'áàäâéèëêíìïîóòöôúùüû','aaaaeeeeiiiioooouuuu'));
 if guess_word is null or guess_word !~ '^[A-ZÑ]{5,7}$' or char_length(guess_word)<>c.word_length or not exists(select 1 from palabreo_private.word_catalog w where w.word=guess_word) then raise exception 'No conozco esa palabra. Prueba con otra.' using errcode='22023';end if;
 select s.answer into strict solution from palabreo_private.solutions s where s.challenge_id=c.id;
 clues:=array_fill('absent'::text,ARRAY[c.word_length::integer]);
 remaining:=string_to_array(solution,null);
 for i in 1..c.word_length loop if substr(guess_word,i,1)=substr(solution,i,1) then clues[i]:='correct';remaining[i]:=null;end if;end loop;
 for i in 1..c.word_length loop
  if clues[i]<>'correct' then j:=array_position(remaining,substr(guess_word,i,1));if j is not null then clues[i]:='present';remaining[j]:=null;end if;end if;
 end loop;
 new_status:=case when guess_word=solution then 'won' when r.attempts=5 then 'lost' else 'playing' end;
 insert into palabreo_private.guesses(round_id,attempt_number,guess,clues,request_id) values(r.id,r.attempts+1,guess_word,to_jsonb(clues),p_request_id);
 update public.competition_rounds set attempts=r.attempts+1,status=new_status,
 finished_at=case when new_status<>'playing' then ended else null end,
 duration_ms=case when new_status<>'playing' then greatest(0,floor(extract(epoch from (ended-r.started_at))*1000))::bigint else null end where id=r.id;
 return palabreo_private.round_view(r.id);
end $function$
;

CREATE OR REPLACE FUNCTION public.palabreo_leagues()
 RETURNS jsonb
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$
begin
 perform palabreo_private.require_scope(null);
 return coalesce((select jsonb_agg(jsonb_build_object('id',l.id,'name',l.name,'length_mode',l.length_mode,'daily_words',l.daily_words,'owner',l.owner_id=auth.uid(),'members',(select count(*) from public.league_members all_members where all_members.league_id=l.id and all_members.left_at is null),'capacity',20) order by l.name) from public.leagues l join public.league_members m on m.league_id=l.id where m.user_id=auth.uid() and m.left_at is null),'[]'::jsonb);
end $function$
;

CREATE OR REPLACE FUNCTION public.palabreo_report_word(p_word text, p_round uuid DEFAULT NULL::uuid)
 RETURNS jsonb
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$
declare v_word text;already boolean;total integer;
begin
 perform palabreo_private.require_scope(null);
 v_word:=upper(translate(lower(trim(p_word)),'áàäâéèëêíìïîóòöôúùüû','aaaaeeeeiiiioooouuuu'));
 if v_word is null or v_word !~ '^[A-ZÑ]{5,7}$' or not exists(select 1 from palabreo_private.word_catalog w where w.word=v_word) then raise exception 'No se encuentra esa palabra';end if;
 if p_round is not null and not exists(select 1 from public.competition_rounds r join palabreo_private.solutions s on s.challenge_id=r.challenge_id where r.id=p_round and r.user_id=auth.uid() and r.status<>'playing' and s.answer=v_word) then raise exception 'Termina tu ronda antes de avisar sobre esta palabra';end if;
 perform 1 from public.profiles where id=auth.uid() for update;
 select exists(select 1 from palabreo_private.word_reports r where r.word=v_word and r.user_id=auth.uid()) into already;
 if not already then
  if (select count(*) from palabreo_private.word_reports where user_id=auth.uid() and created_at>clock_timestamp()-interval '24 hours')>=10 then raise exception 'Puedes avisar de hasta diez palabras diferentes cada 24 horas';end if;
  insert into palabreo_private.word_reports(word,user_id) values(v_word,auth.uid());
 end if;
 select count(*)::integer into total from palabreo_private.word_reports r where r.word=v_word;
 return jsonb_build_object('already_reported',already,'reports',total,'review_needed',total>=3);
end $function$
;

BEGIN
 SELECT c.word_length INTO STRICT expected FROM public.competition_rounds r
 JOIN public.challenges c ON c.id=r.challenge_id WHERE r.id=NEW.round_id;
 IF char_length(NEW.guess)<>expected THEN RAISE EXCEPTION 'Longitud de intento incorrecta' USING ERRCODE='22023';END IF;
 RETURN NEW;
END $body$;
